% Boxed text environment
\newsavebox{\boxedtextbox}

\newenvironment{boxedtext}
  {%
    \par\noindent
    \begin{lrbox}{\boxedtextbox}%
      \begin{minipage}{\dimexpr\textwidth-2\fboxsep-2\fboxrule\relax}
      \ignorespaces
  }
  {%
      \end{minipage}%
    \end{lrbox}%
    \fbox{\usebox{\boxedtextbox}}%
    \par
  }

% Boxed text that is single item
\newenvironment{boxeditem}[1][\relax]{%
    \begin{boxedtext}
    \begin{itemize}
      \ifx#1\relax % Use dot if no argument given
        \item
      \else
        \item[#1]
      \fi
      \ignorespaces}
  {%
    \end{itemize}
    \end{boxedtext}
  }
